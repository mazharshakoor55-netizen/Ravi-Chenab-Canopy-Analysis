// Mapping and Analyzing Canopy Height Using Global CHM Dataset in Google Earth Engine: A Case Study of a Ravi-Chenab Interfluve sub-basin

/*                                             // 1. Canopy height //
Canopy height refers to the height of trees above the ground topography. 
It is a measurement used in various applications, including forestry, ecology, and environmental studies.

                                           // 2. Canopy Height Model (CHM) //
The Canopy Height Model (CHM) dataset is a high-resolution (1 meter) raster layer that maps the tree height as a continuous surface. 
Each pixel in the CHM represents the tree overstory height above the underlying ground topography. 
The key data source for creating a CHM is LIDAR data. LIDAR (Light Detection and Ranging) technology enables accurate measurement of elevation 
for different ground surface types based on the light pulses reflected from the earth. 
LIDAR systems produce large volumes of 3D point cloud data which can be analyzed to map different surface types and their elevation. 
The accuracy and quality of a CHM improves with higher point density LIDAR data.
*/

// 1. Load the HydroBASINS level 6 dataset to define the region of interest (ROI)
var watershed = ee.FeatureCollection("WWF/HydroSHEDS/v1/Basins/hybas_6");
Map.addLayer(watershed, [], 'Watershed');

// Loads the HydroBASINS level 6 dataset and adds it to the map for visualization.
// 2. Define the coordinates of a point within the ROI
var coordinates = [74.36269411022569, 32.006944438400176];
var pointOfInterest = ee.Geometry.Point(coordinates);
// 3. Filter the basin collection to get the specific ROI containing the point
var regionOfInterest = watershed.filterBounds(pointOfInterest);

// 4. Center the map on the ROI and add it to the map
Map.centerObject(regionOfInterest);
Map.addLayer(regionOfInterest, [], 'Study Area');

// Defines a point of interest and filters the basin collection to get the specific ROI containing this point. 
// The ROI is then centered on the map and added for visualization

// 5. Load the canopy height image collection and create a mosaic // (Loading and Visualizing Canopy Height Data)
var canopyHeightCollection = ee.ImageCollection("projects/meta-forest-monitoring-okw37/assets/CanopyHeight").mosaic();
print(canopyHeightCollection,'Canopy Height Collection Properties');

// 6. Define visualization parameters for the canopy height image
var visParams1 = {
  "opacity": 1, // Controls the transparency of the image. An opacity of 1 means the image is fully opaque
  "bands": ["cover_code", "cover_code", "cover_code"], // Uses the "cover_code" band for red, green, and blue channels, resulting in a grayscale visualization.
  "gamma": 1 // No adjustment to brightness and contrast
};

var visParams2 = {
  "opacity": 1,
  "bands": ["cover_code", "cover_code", "cover_code"],
  "min": 1.6269897434224054, // Sets the minimum value for the data scaling
  "max": 10.22118610067458,  // Sets the maximum value for the data scaling
  "gamma": 1
};

// 7. Add the canopy height image, clipped to the ROI, to the map
Map.addLayer(canopyHeightCollection.clip(regionOfInterest), visParams2, 'Canopy Height', false);
print(ui.Chart.image.histogram(canopyHeightCollection, regionOfInterest, 100));

// Loads the canopy height image collection, creates a mosaic, and defines visualization parameters. 
// The canopy height image is then clipped to the ROI and added to the map.

// 8. Create a mask for forest areas with canopy height greater than 0 (Creating a Forest Mask and Visualizing Forest Areas)
var forestMask = canopyHeightCollection.gt(0);
var forestAreas = canopyHeightCollection.updateMask(forestMask);

// 9. Add the masked forest areas to the map
Map.addLayer(forestAreas.clip(regionOfInterest), [], 'Forest Areas', false);

// Creates a mask for forest areas with canopy height greater than 0, updates the mask on the canopy height image, 
// and visualizes the forest areas on the map.

// 10. Display a histogram of canopy height values within the ROI // (Displaying Histogram and Calculating Canopy Height Statistics)
print(ui.Chart.image.histogram(forestAreas, regionOfInterest, 100));

// 11. Calculate and print the mean canopy height within the ROI
var meanCanopyHeight = forestAreas.reduceRegion({
  reducer: ee.Reducer.mean(),
  geometry: regionOfInterest,
  scale: 100
}).values().get(0);
print('Mean Canopy Height:', meanCanopyHeight);

// 12. Calculate and print the maximum canopy height within the ROI
var maxCanopyHeight = forestAreas.reduceRegion({
  reducer: ee.Reducer.max(),
  geometry: regionOfInterest,
  scale: 100
}).values().get(0);
print('Max Canopy Height:', maxCanopyHeight);

// 13. Calculate and print the minimum canopy height within the ROI
var minCanopyHeight = forestAreas.reduceRegion({
  reducer: ee.Reducer.min(),
  geometry: regionOfInterest,
  scale: 100
}).values().get(0);
print('Min Canopy Height:', minCanopyHeight);

// 14. Calculate and print the 99th percentile canopy height within the ROI
var canopyHeight99thPercentile = forestAreas.reduceRegion({
  reducer: ee.Reducer.percentile([99]),
  geometry: regionOfInterest,
  scale: 100
}).values().get(0);
print('Percentile Canopy Height:', canopyHeight99thPercentile);

// Displays a histogram of canopy height values within the ROI. 
// Calculates and prints the mean, maximum, and 99th percentile canopy height within the ROI.

// 15. Create a constant image with value 0 for Canopy Height Classification (Classifying Canopy Height and Visualizing)
var baseImage = ee.Image.constant(0);

// 16. Classify the forest areas into four classes based on canopy height
var class1 = baseImage.where(forestAreas.lt(5), 1);
Map.addLayer(class1.clip(regionOfInterest), [], 'Canopy Height 0 to 5 meters', false);
// class1: Canopy Height ranging from 0 to 5 meters

var class2 = class1.where(forestAreas.gte(5).and(forestAreas.lt(10)), 2);
Map.addLayer(class2.clip(regionOfInterest), [], 'Canopy Height 5 to 10 meters', false);
// class2: Canopy Height ranging from 5 to 10 meters

var class3 = class2.where(forestAreas.gte(10).and(forestAreas.lt(20)), 3);
Map.addLayer(class3.clip(regionOfInterest), [], 'Canopy Height 10 to 20 meters', false);
// class3: Canopy Height ranging from 10 to 20 meters

var class4 = class3.where(forestAreas.gte(20), 4);
Map.addLayer(class4.clip(regionOfInterest), [], 'Canopy Height greater than 20 meters', false);
// class4: Canopy Height greater than 20 meters

// 17. Add the classified canopy height image to the map with color palette
Map.addLayer(class4.clip(regionOfInterest), {
  min: 0,
  max: 4,
  palette: ['black', 'red', 'orange', 'green', 'blue']
}, 'Classified Canopy Height', false);

// Creates a constant image and classifies the forest areas into four classes based on canopy height. 
// The classified image is then visualized on the map.
// ==========================================
// =========================================================================
// 18. Create and Add the Interactive Map Legend
// =========================================================================

// 1. Set up the main panel container for the legend
var legend = ui.Panel({
  style: {
    position: 'bottom-left',   // Places the panel at the bottom-left of the map view
    padding: '8px 15px',
    backgroundColor: 'white',
    border: '1px solid ccc'
  }
});

// 2. Create and style the legend title
var legendTitle = ui.Label({
  value: 'Canopy Height Classes',
  style: {
    fontWeight: 'bold',
    fontSize: '16px',
    margin: '0 0 8px 0',
    padding: '0',
    color: '333333'
  }
});

// Add the title to our legend box
legend.add(legendTitle);

// 3. Define a reusable function to build a legend row (Color Box + Text Description)
var makeLegendRow = function(color, description) {
  
  // This creates the small square color indicator
  var colorBox = ui.Label({
    style: {
      backgroundColor: color,
      padding: '8px',          // Controls the height and width of the square box
      margin: '0 8px 4px 0',
      border: '0.5px solid #888'
    }
  });

  // This creates the descriptive label text next to the color box
  var textLabel = ui.Label({
    value: description,
    style: {
      margin: '0 0 4px 0',
      fontSize: '13px',
      color: '444444'
    }
  });

  // Returns a horizontal row containing both elements side-by-side
  return ui.Panel({
    widgets: [colorBox, textLabel],
    layout: ui.Panel.Layout.Flow('horizontal')
  });
};

// 4. Define your color palette and class description text matching step #17
// Palette maps directly to Class 1, Class 2, Class 3, and Class 4 
var paletteColors = ['red', 'orange', 'green', 'blue'];
var classNames = [
  '0 to 5 meters (Low Canopy)',
  '5 to 10 meters (Medium Canopy)',
  '10 to 20 meters (High Canopy)',
  'Greater than 20 meters (Dense Forest Overstory)'
];

// 5. Loop through the arrays to build and add each row to the panel
for (var i = 0; i < 4; i++) {
  legend.add(makeLegendRow(paletteColors[i], classNames[i]));
}

// 6. Project the completed legend onto the active map console interface
Map.add(legend);
